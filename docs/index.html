Powerful Cross-Environment Engine.

The compatibility engine is written in an optimized, memory-safe Rust library called Shinkore.

It is compiled to WebAssembly using wasm-bindgen, allowing for use of the engine in both Browser and Node.js environments.

Node.js Code

import { SKEngine } from 'shinkom/engine';

const mockHTML = `
    <div style="display: grid; gap: 20px;">
        <dialog open>
            <p>Testing native dialog elements</p>
        </dialog>
        <picture>
            <source srcset="image.webp" type="image/webp">
            <img src="image.jpg" alt="Fallback">
        </picture>
    </div>
`

async function runAnalysis(html) {
    const engine = new SKEngine()

    try {
        await engine.initEngine()

        const depthLevel = 3
        const result = engine.checkElements(html, depthLevel)

        if (!result) {
            console.log("No results returned")
        }
    } catch(error) {
        console.error(`Run failed: ${error}`)
    } finally {
        engine.destroy()
    }
}

runAnalysis(mockHTML)
